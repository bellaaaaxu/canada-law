# M2（联邦法规）实施计划

> 依据 `SPEC.md`「联邦（Justice Laws）」（实测事实）和「联邦部分怎么做」（2026-09-26 定的设计）。同一会话内按任务顺序做 TDD：先写会失败的测试 → 跑一遍确认失败 → 实现 → 跑通 → 提交。在本地分支 `m2-federal` 上提交，合并和推送前先问维护者。
> BC 部分的行为一律不变：每个任务结束时 `npm test` 全绿（现有 175 条照样通过）。

**目标：** 5 个工具、skill 小程序、MCP 单文件和 `.mcpb` 都能查联邦法规（Canada Labour Code 及其条例），引用契约不变，发布 v0.2.0。

**做法：** 和 BC 一样实时取 Justice Laws 官网的 XML 和网页，缓存 24 小时；联邦搜索在内存里做（法典加名下条例，名单来自官方清单 `Legis.xml`）；不装任何原生组件。

**技术：** TypeScript、fast-xml-parser（已有）、vitest（已有）、esbuild 打包（已有）。

## 接口（后面各任务按这个写，名字不要变）

```ts
// src/types.ts
export type ActCandidate = { act_id: string; title: string; citation: string; type: 'act' | 'regulation'; source_url: string; status?: 'current' | 'repealed or replaced' | 'unknown'; note?: string };
export type SearchResult = Citation & { snippet: string; match: string[] };

// src/sources/xml.ts
export function cutSnippet(body: string, whenNoHit: () => string): string; // body 里命中词用 ** 包着；规则同 BC（600 字内整条，否则按分句）

// src/sources/federal-meta.ts
export const FED_BASE = 'https://laws-lois.justice.gc.ca';
export type FedPageMeta = { title: string | null; citation: string | null; currentTo: string | null; lastAmended: string | null };
export function parseFedPage(html: string): FedPageMeta;
export type LegisEntry = { id: string; uniqueId: string; kind: 'act' | 'regulation'; title: string; officialNumber: string | null; ref: string | null; regRefs: string[] };
export function parseLegis(xml: string): LegisEntry[];          // 只要英文；id 取 LinkToXML 里的文件名
export function citationFromLegis(e: LegisEntry): string;       // "R.S.C., 1985, c. L-2" / "S.C. 1996, c. 23" / "C.R.C., c. 986" / "SOR/86-304"
export function normalizeFedId(input: string): string;          // "C.R.C., c. 986" → "C.R.C.,_c._986"；"SOR/86-304" → "SOR-86-304"；不合法抛 ToolError
export function isFederalId(id: string): boolean;               // 小程序用：带 "-" 或以 "C.R.C." 开头

// src/sources/federal-xml.ts
export type FedDocInfo = { kind: 'act' | 'regulation' | 'unknown'; title: string | null; instrumentNumber: string | null; lastAmended: string | null; allRepealed: boolean };
export function fedDocInfo(doc: XNode): FedDocInfo;
export type FedSection = { node: XNode; num: string; heading: string | null; location: string[]; nearestHeading: string | null; repealed: boolean; range: boolean };
export function bodySections(doc: XNode): FedSection[];         // 只取 <Body> 里的条，按原顺序
export function findFedSection(sections: FedSection[], num: string): FedSection | null; // 精确条号，或落在 "163 to 165" / "5.10 and 5.11" 里
export function renderFedSection(section: XNode, o?: { mark?: RegExp }): string; // mark：把命中词包成 **…**（搜索摘要用）
export function hasNotInForcePart(section: XNode): boolean;     // in-force="no"
export type NotInForce = { citation: string; sections: string[] };
export function notInForce(doc: XNode): NotInForce[];           // AMENDMENTS NOT IN FORCE：新加的条号 + 被替换/修改/废止的条号
export function fedToc(doc: XNode): string;
export type SectionRecord = { num: string; heading: string | null; nearestHeading: string | null; location: string[]; text: string; notes: string; definedTerms: string[]; node: XNode };
export function sectionRecords(doc: XNode): SectionRecord[];    // 搜索用；不含范围条号和已废止的条

// src/sources/federal.ts
export class FederalClient {
  constructor(opts: { fetcher: Fetcher });
  findAct(name: string): Promise<ActCandidate[]>;
  getToc(actId: string): Promise<{ act: Omit<Citation, 'section' | 'heading'>; outline: string; warnings: string[]; notice: string }>;
  getSection(actId: string, section: string): Promise<{ citation: Citation; location: string[]; text: string; warnings: string[]; notice: string }>;
  search(query: string, limit?: number): Promise<{ query: string; documents_searched: number; results: SearchResult[]; warnings: string[]; notes: string[]; notice: string }>;
}
```

网址：XML `${FED_BASE}/eng/XML/{id}.xml`；目录页 `${FED_BASE}/eng/{acts|regulations}/{id}/index.html`；条级页 `…/section-{n}.html`；清单 `${FED_BASE}/eng/XML/Legis.xml`。法还是条例看 XML 根元素（`Statute` / `Regulation`）。
